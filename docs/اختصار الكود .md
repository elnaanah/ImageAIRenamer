using System;
using System.IO;
using System.Net.Http;
using System.Text;
using System.Threading.Tasks;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

namespace GeminiImageAnalyzer
{
    // تعريف كائن للرد المتوقع بصيغة JSON
    public class AnalysisResult
    {
        public bool IsFound { get; set; }
        public string Description { get; set; }
        public double Confidence { get; set; }
    }

    class Program
    {
        // ⚠️ استبدل هذا بمفتاح الـ API الخاص بك
        private const string ApiKey = "YOUR_GOOGLE_API_KEY_HERE";
        
        // استخدام نموذج Gemini 1.5 Flash لأنه سريع ويدعم الصور بشكل ممتاز
        private const string ApiUrl = "https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent";

        static async Task Main(string[] args)
        {
            Console.OutputEncoding = Encoding.UTF8;
            Console.WriteLine("--- Gemini Image Analyzer ---");

            // 1. تحديد مسار الصورة والشيء الذي نبحث عنه
            string imagePath = "test_image.jpg"; // تأكد من وجود صورة بهذا الاسم بجانب ملف التشغيل
            string searchItem = "قطة"; // الشيء الذي نبحث عنه

            if (!File.Exists(imagePath))
            {
                Console.WriteLine($"خطأ: لم يتم العثور على الصورة في المسار: {imagePath}");
                return;
            }

            try
            {
                Console.WriteLine("جاري تحليل الصورة...");
                string resultJson = await AnalyzeImage(imagePath, searchItem);

                // تنظيف الرد (لإزالة علامات Markdown مثل ```json)
                resultJson = CleanJsonString(resultJson);

                // تحويل النص إلى كائن
                var analysis = JsonConvert.DeserializeObject<AnalysisResult>(resultJson);

                Console.WriteLine("\n--- النتيجة ---");
                Console.WriteLine($"هل '{searchItem}' موجود؟: {(analysis.IsFound ? "نعم ✅" : "لا ❌")}");
                Console.WriteLine($"الوصف: {analysis.Description}");
                Console.WriteLine($"نسبة الثقة: {analysis.Confidence}%");
                Console.WriteLine("----------------");
                Console.WriteLine("JSON الخام:\n" + resultJson);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"حدث خطأ: {ex.Message}");
            }
            
            Console.ReadKey();
        }

        static async Task<string> AnalyzeImage(string imagePath, string searchItem)
        {
            // قراءة الصورة وتحويلها إلى Base64
            byte[] imageBytes = await File.ReadAllBytesAsync(imagePath);
            string base64Image = Convert.ToBase64String(imageBytes);

            // بناء الـ Prompt لضمان الحصول على JSON دقيق
            string promptText = $@"
            انظر إلى هذه الصورة. هل يوجد '{searchItem}' فيها؟
            أجب فقط بصيغة JSON وبدون أي نصوص إضافية، اتبع الهيكل التالي:
            {{
                ""IsFound"": true/false,
                ""Description"": ""وصف قصير لما وجدته يتعلق بالبحث"",
                ""Confidence"": 95.5
            }}
            ";

            using (var client = new HttpClient())
            {
                // بناء هيكل الطلب حسب توثيق Google Gemini REST API
                var requestBody = new
                {
                    contents = new[]
                    {
                        new
                        {
                            parts = new object[]
                            {
                                new { text = promptText },
                                new
                                {
                                    inline_data = new
                                    {
                                        mime_type = "image/jpeg", // عدل هذا إذا كانت الصورة PNG
                                        data = base64Image
                                    }
                                }
                            }
                        }
                    },
                    // إعدادات لتقليل الإبداع وزيادة الدقة
                    generationConfig = new
                    {
                        temperature = 0.1,
                        responseMimeType = "application/json" // إجبار النموذج على الرد بـ JSON
                    }
                };

                string jsonContent = JsonConvert.SerializeObject(requestBody);
                var httpContent = new StringContent(jsonContent, Encoding.UTF8, "application/json");

                // إرسال الطلب
                string urlWithKey = $"{ApiUrl}?key={ApiKey}";
                HttpResponseMessage response = await client.PostAsync(urlWithKey, httpContent);

                string responseString = await response.Content.ReadAsStringAsync();

                if (!response.IsSuccessStatusCode)
                {
                    throw new Exception($"API Error: {response.StatusCode} - {responseString}");
                }

                // استخراج النص من رد Gemini المعقد
                var jsonResponse = JObject.Parse(responseString);
                string textResult = jsonResponse["candidates"]??["content"]?["parts"]??["text"]?.ToString();

                return textResult;
            }
        }

        // دالة مساعدة لتنظيف كود الـ JSON من أي تنسيق إضافي يضعه الذكاء الاصطناعي
        static string CleanJsonString(string json)
        {
            if (string.IsNullOrEmpty(json)) return json;
            
            // إزالة ```json و ``` إذا كانت موجودة
            json = json.Replace("```json", "").Replace("```", "").Trim();
            return json;
        }
    }
}
